% ---------------------------------------------------------------------------
% Textbook density.
% The book previously used parskip=half with a 4.80in measure, which reads
% loose and airy. A textbook wants indented paragraphs, a wider measure, and
% tighter leading. Vertical space is reserved for structure, not for every
% paragraph break.
% ---------------------------------------------------------------------------
\setlength{\parskip}{0pt plus 0.6pt}
\setlength{\parindent}{1.2em}

% Lists sit closer to the text they belong to.
\setlist[itemize]{leftmargin=1.2em,topsep=3pt,itemsep=1.5pt,parsep=0pt}
\setlist[enumerate]{leftmargin=1.5em,topsep=3pt,itemsep=1.5pt,parsep=0pt}

% Tables carry a lot of this book's argument. Give the rows air without
% the balloon that arraystretch 1.25 produced at the old measure.
\renewcommand{\arraystretch}{1.14}
\setlength{\tabcolsep}{5pt}

% ---------------------------------------------------------------------------
% Wide tables and figures may use the margin column.
% The co-design matrix is five columns wide and is illegible inside a 5.30in
% measure. `fullwidth` spans text plus margin, which is the natural home for
% any artifact the reader is meant to scan across.
% ---------------------------------------------------------------------------
\usepackage{environ}
\newlength{\fullwidthlen}
\setlength{\fullwidthlen}{\dimexpr\textwidth+\marginparsep+\marginparwidth\relax}
\NewEnviron{fullwidth}{%
  \par\noindent
  \makebox[\textwidth][l]{%
    \begin{minipage}{\fullwidthlen}\BODY\end{minipage}%
  }\par
}

% Quarto emits .column-page / .column-body-outset divs for wide content.
\providecommand{\columnpage}[1]{\begin{fullwidth}#1\end{fullwidth}}

% Hyphenation: a narrow measure with long technical compounds produces a
% ladder of hyphens. Loosen the penalty rather than the justification.
\hyphenpenalty=180
\tolerance=1400
\emergencystretch=2.2em
