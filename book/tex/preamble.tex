% The two-world-state motif. Proposal remains dashed until an allowed action
% changes the world from W_t to W_{t+1}.
\newcommand{\patransition}{%
  \begin{tikzpicture}[baseline=-0.6ex]
    \node[draw=white,text=white,rounded corners=1.5pt,minimum width=0.78cm,minimum height=0.46cm,font=\sffamily\bfseries\scriptsize] (wt) {$W_t$};
    \node[right=0.34cm of wt,text=proposal,font=\sffamily\scriptsize] (p) {proposal};
    \node[right=0.34cm of p,draw=measured,text=white,fill=measured,minimum width=0.16cm,minimum height=0.42cm] (g) {};
    \node[right=0.34cm of g,draw=white,text=white,rounded corners=1.5pt,minimum width=0.92cm,minimum height=0.46cm,font=\sffamily\bfseries\scriptsize] (wtone) {$W_{t+1}$};
    \draw[white,line width=0.8pt,-{Stealth[length=1.8mm]}] (wt) -- (p);
    \draw[proposal,dashed,line width=1.1pt,-{Stealth[length=1.8mm]}] (p) -- (g);
    \draw[measured,line width=1.2pt,-{Stealth[length=1.8mm]}] (g) -- (wtone);
  \end{tikzpicture}}
