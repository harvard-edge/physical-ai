% Physical AI — PDF style. LuaLaTeX + KOMA scrbook, Quarto-safe.
% (Quarto loads fontspec, xcolor, then hyperref AFTER this include.)

\usepackage{microtype}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}
\usepackage{scrlayer-scrpage}
\usepackage{needspace}

% Brand palette.
\definecolor{reflex}{HTML}{18A999}
\definecolor{deliberation}{HTML}{1B2A4A}
\definecolor{learning}{HTML}{E8A33D}
\definecolor{ink}{HTML}{111318}
\definecolor{softink}{HTML}{404856}
\definecolor{paleink}{HTML}{697386}

% Links: deferred so hyperref is loaded first.
\AtBeginDocument{%
  \hypersetup{colorlinks=true, linkcolor=deliberation, urlcolor=reflex,
    citecolor=reflex, filecolor=reflex}%
}

% Body rhythm.
\frenchspacing
\setlength{\emergencystretch}{2em}
\microtypesetup{protrusion=true, expansion=true, final}

% Headings: Avenir Next (sansfont), indigo, restrained.
\setkomafont{disposition}{\normalfont\sffamily\color{deliberation}}
\setkomafont{chapter}{\normalfont\sffamily\bfseries\fontsize{26}{30}\selectfont\color{deliberation}}
\setkomafont{section}{\normalfont\sffamily\bfseries\fontsize{16}{20}\selectfont\color{deliberation}}
\setkomafont{subsection}{\normalfont\sffamily\bfseries\fontsize{12.5}{16}\selectfont\color{deliberation}}
\setkomafont{captionlabel}{\normalfont\sffamily\bfseries\color{deliberation}}
\setkomafont{caption}{\normalfont\small\color{softink}}

% Chapter opener: a small amber label above the title, then a subtle rule.
\renewcommand*{\chapterformat}{%
  {\normalfont\sffamily\bfseries\fontsize{11}{13}\selectfont\color{learning}CHAPTER~\thechapter}\par
  \vspace{0.45\baselineskip}}
\renewcommand*{\chapterlinesformat}[3]{%
  \vspace*{0.2in}%
  #2#3\par\nobreak
  \vspace{0.55\baselineskip}{\textcolor{softink}{\rule{0.4\linewidth}{0.4pt}}}}

% Section spacing.
\RedeclareSectionCommand[beforeskip=1.6\baselineskip,afterskip=0.5\baselineskip]{section}
\RedeclareSectionCommand[beforeskip=1.2\baselineskip,afterskip=0.35\baselineskip]{subsection}

% Running heads/feet: chapter mark inner, page number outer, teal hairline.
\clearpairofpagestyles
\automark[chapter]{chapter}
\renewcommand*{\chaptermarkformat}{}
\setkomafont{pagehead}{\normalfont\sffamily\small\color{softink}}
\setkomafont{pagenumber}{\normalfont\sffamily\small\color{softink}}
\ihead{\headmark}
\ohead{\pagemark}
\KOMAoptions{headsepline=0.3pt}
\addtokomafont{headsepline}{\color{paleink}}

% TOC: chapters indigo bold.
\setkomafont{chapterentry}{\normalfont\sffamily\bfseries\color{deliberation}}
\setkomafont{chapterentrypagenumber}{\normalfont\sffamily\bfseries\color{deliberation}}

% Title-page fonts.
\setkomafont{title}{\normalfont\sffamily\bfseries\fontsize{30}{34}\selectfont\color{deliberation}}
\setkomafont{subtitle}{\normalfont\sffamily\fontsize{13}{17}\selectfont\color{softink}}
\setkomafont{author}{\normalfont\large\color{ink}}
\setkomafont{date}{\normalfont\normalsize\color{softink}}

% The closed-loop motif, drawn in TikZ (no external image).
\newcommand{\paloop}[1]{%
  \begin{tikzpicture}
    \draw[reflex,line width=1.4pt,-{Stealth[length=2.4mm]}] (0:#1) arc (0:312:#1);
    \fill[learning] (312:#1) circle (0.13*#1);
  \end{tikzpicture}}
